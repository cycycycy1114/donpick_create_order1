let productList = []; // 1. 從 A 資料庫讀取商品清單 async function loadProducts() { try { const res = await fetch('/api/get-products'); productList = await res.json(); const select = document.getElementById('productSelect'); if (!Array.isArray(productList) || productList.length === 0) { select.innerHTML = ''; return; } select.innerHTML = ''; productList.forEach((p, idx) => { const opt = document.createElement('option'); opt.value = idx; opt.textContent = `${p.name} ($${p.price})`; select.appendChild(opt); }); } catch (e) { document.getElementById('productSelect').innerHTML = ''; } } // 當選擇商品改變時，自動更新「規格選項」與「價格」 function onProductChange() { const idx = document.getElementById('productSelect').value; const specSelect = document.getElementById('specSelect'); specSelect.innerHTML = ''; // 清空舊規格 if (idx !== "" && productList[idx]) { const product = productList[idx]; // 帶入該商品的規格 if (product.specs && product.specs.length > 0) { product.specs.forEach(s => { const opt = document.createElement('option'); opt.value = s; opt.textContent = s; specSelect.appendChild(opt); }); } else { specSelect.innerHTML = ''; } } else { specSelect.innerHTML = ''; } updatePrice(); } // 2. 自動即時計算價格 function updatePrice() { const idx = document.getElementById('productSelect').value; const qty = parseInt(document.getElementById('quantity').value) || 1; if (idx !== "" && productList[idx]) { const price = productList[idx].price || 0; document.getElementById('unitPrice').innerText = price; document.getElementById('totalPrice').innerText = price * qty; } else { document.getElementById('unitPrice').innerText = '0'; document.getElementById('totalPrice').innerText = '0'; } } document.getElementById('productSelect').addEventListener('change', onProductChange); document.getElementById('quantity').addEventListener('input', updatePrice); loadProducts();
